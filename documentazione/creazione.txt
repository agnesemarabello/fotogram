/* CREAZIONE TABELLE */
CREATE TABLE UTENTE (
    username VARCHAR(25) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(50) NOT NULL,
    timestamp_user TIMESTAMP NOT NULL,
    immagine_profilo BYTEA NOT NULL,
    versione_immagine INTEGER NOT NULL DEFAULT 0,
    ruolo VARCHAR(25) NOT NULL CHECK (ruolo IN ('admin', 'moderatore', 'utente')),
    limitato BOOLEAN DEFAULT false,
    PRIMARY KEY (username),
    CHECK (LENGTH(immagine_profilo) <= 102400)
);
CREATE TABLE FOLLOWING_LIST (
    username_seguito VARCHAR(25) NOT NULL,
    username_seguace VARCHAR(25) NOT NULL,
    timestamp_following TIMESTAMP NOT NULL,
    PRIMARY KEY (username_seguito, username_seguace),
    FOREIGN KEY (username_seguito) REFERENCES UTENTE (username),
    FOREIGN KEY (username_seguace) REFERENCES UTENTE (username)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION
);
CREATE TABLE SESSIONE (
    username VARCHAR(25) NOT NULL,
    access_token TEXT UNIQUE NOT NULL,
    timestamp_inizio TIMESTAMP NOT NULL,
    timestamp_fine TIMESTAMP,
    refresh_token TEXT UNIQUE NOT NULL,
    refresh_scadenza TIMESTAMP NOT NULL,
    PRIMARY KEY (username, access_token),
    FOREIGN KEY (username) REFERENCES UTENTE (username)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION
);
CREATE TABLE ADMIN (
    username VARCHAR(25) NOT NULL,
    PRIMARY KEY (username),
    FOREIGN KEY (username) REFERENCES UTENTE (username)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION
);
CREATE TABLE MODERATORE (
    username VARCHAR(25) NOT NULL,
    timestamp_nomina TIMESTAMP NOT NULL,
    ADMIN_username VARCHAR(25) NOT NULL,
    PRIMARY KEY (username),
    FOREIGN KEY (username) REFERENCES UTENTE (username)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION,
    FOREIGN KEY (ADMIN_username) REFERENCES ADMIN (username)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION
);
CREATE TABLE POST (
    post_id INTEGER NOT NULL,
    timestamp_post TIMESTAMP NOT NULL,
    moderato BOOLEAN DEFAULT false NOT NULL,
    flaggato BOOLEAN DEFAULT false NOT NULL,
    testo TEXT,
    immagine BYTEA,
    autore VARCHAR(25) NOT NULL,
    PRIMARY KEY (post_id),
    FOREIGN KEY (autore) REFERENCES UTENTE (username)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION,
    
    CHECK (NOT (moderato = TRUE AND flaggato = TRUE)),
    CHECK (LENGTH(immagine) <= 102400)
);
CREATE TABLE MI_PIACE (
    username VARCHAR(25) NOT NULL,
    post_id INTEGER NOT NULL,
    timestamp_like TIMESTAMP NOT NULL,
    PRIMARY KEY (username, post_id),
    FOREIGN KEY (username) REFERENCES UTENTE (username)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION,
    FOREIGN KEY (post_id) REFERENCES POST (post_id)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION
);
CREATE TABLE FLAG (
    username VARCHAR(25) NOT NULL,
    post_id INTEGER NOT NULL,
    motivazione TEXT,
    timestamp_flag TIMESTAMP NOT NULL,
    PRIMARY KEY (username, post_id),
    FOREIGN KEY (username) REFERENCES UTENTE (username)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION,
    FOREIGN KEY (post_id) REFERENCES POST (post_id)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION
);
CREATE TABLE MODERAZIONE (
    post_id INTEGER NOT NULL,
    timestamp_moderazione TIMESTAMP NOT NULL,
    ADMIN_username VARCHAR(25),
    MODERATORE_username VARCHAR(25),
    PRIMARY KEY (post_id),
    FOREIGN KEY (post_id) REFERENCES POST (post_id)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION,
    FOREIGN KEY (ADMIN_username) REFERENCES ADMIN (username)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION,
    FOREIGN KEY (MODERATORE_username) REFERENCES MODERATORE (username)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION,

    CHECK (
        (ADMIN_username IS NOT NULL AND MODERATORE_username IS NULL)
        OR
        (ADMIN_username IS NULL AND MODERATORE_username IS NOT NULL)
    )
);
